### 第3步：在代码中使用（1分钟）

```cpp
// MainInterfaceWidget.cpp

#include "Registration2D3DService.h"

void MainInterfaceWidget::testRegistration()
{
    // 1. 获取服务
    ctkServiceReference ref = 
        m_pluginContext->getServiceReference<Registration2D3DService>();
    
    Registration2D3DService* service = 
        m_pluginContext->getService<Registration2D3DService>(ref);
    
    if (!service) {
        qWarning() << "配准服务不可用";
        return;
    }
    
    qDebug() << "✅ 配准服务已连接";
    
    // 2. 初始化Python环境
    QString appDir = QCoreApplication::applicationDirPath();
    QString scriptsPath = appDir + "/Regi";
    
    bool pythonOk = service->initializePythonEnvironment("", scriptsPath);
    if (pythonOk) {
        qDebug() << "✅ Python环境初始化成功";
    } else {
        qWarning() << "❌ Python初始化失败:" << service->getLastError();
        return;
    }
    
    // 3. 准备配准参数
    Registration2D3DParameters params;
    params.ctPath = "D:/data/patient001/ct.nrrd";
    params.xrayApPath = "D:/data/patient001/xray_ap.png";
    params.xrayLatPath = "D:/data/patient001/xray_lat.png";
    params.initParams = {0, 0, 0, 0, 0, 0};
    params.searchRange = {15, 15, 15, 50, 50, 50};
    params.kdTreeNum = 50;
    
    // 4. 连接信号
    connect(service, &Registration2D3DService::progressUpdated,
            [](const QString& id, const Registration2D3DProgress& progress) {
        qDebug() << "进度:" << progress.percentage << "%";
    });
    
    connect(service, &Registration2D3DService::registrationCompleted,
            [](const QString& id, const Registration2D3DResult& result) {
        qDebug() << "✅ 配准完成!";
        qDebug() << "AP: rx=" << result.apResult.rx 
                 << "ry=" << result.apResult.ry;
        qDebug() << "耗时:" << result.durationSeconds << "秒";
    });
    
    // 5. 启动配准
    QString regId = service->startRegistration(params);
    qDebug() << "配准任务ID:" << regId;
}
```

**运行应用**：
```bash
cd Release
medicalpro.exe
```

## 💡 提示

### 性能优化
- K-d树数量越大，配准越精确，但耗时越长
- 推荐值：快速测试用30，正式使用用50-100

### 参数调整
- 初始参数越接近真实值，配准越快
- 搜索范围要覆盖可能的误差范围

### GPU加速
- 确保安装了CUDA和最新的NVIDIA驱动
- 检查PyTorch是否支持GPU：`python -c "import torch; print(torch.cuda.is_available())"`
